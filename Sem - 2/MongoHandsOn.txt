=Viewing Data in Collection=
=db.collectionname.find()
->db.Cars.find()

->db.Cars.find({},{model:1})//{} 1st wala jo hai wo where condition ke liye hai ;second position wala * jo hota hai select statement mein uske liye hai aur ye jo select kar rahe hai usme 0 , 1 , -1 hota hai 0 nahi ke liye , 1 true , -1 ek order ke liye;

->db.Cars.find({fuel_type:"petrol"})

->db.Cars.find({"engine.type":"Turbocharged"});//in case of using nested object you need to use "" to access inner data


=db.collectionname.findOne()

->db.Cars.findOne()


=Upating Document/Row=
=db.collectionname.updateOne({condition},{what to set/push , also need to use $set to access the key})

->db.Cars.updateOne(
	{"model":"Nexon"},
	{$set : {color:"red"}}
)

->db.Cars.updateOne(
	{model:"nexon"},
	{$set : {color:"red"}}//maybe set is for adding 
)

->db.Cars.updateOne(
	{model:nexon"},
	{$push : {features:"Heated Seats"}}//maybe push is for concat
)

->db.Cars.updateOne(
	{model:"nexon"},
	{$set : {features:"Set Try Kiya"}}
)
